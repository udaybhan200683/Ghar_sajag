cmake_minimum_required(VERSION 3.16)

option(GS_HIL_BUILD "Enable the USB HIL control plane (never for release images)" OFF)
option(GS_HIL_CONTROL "Enable test-only UART controls while retaining product security/runtime paths" OFF)
option(GS_HIL_FORCE_BOOT_HEALTH_FAIL "Force the signed HIL candidate to fail OTA boot health for rollback qualification" OFF)
if(GS_HIL_BUILD)
    set(GS_HIL_CONTROL ON)
endif()
if(GS_HIL_FORCE_BOOT_HEALTH_FAIL AND NOT GS_HIL_CONTROL)
    message(FATAL_ERROR "GS_HIL_FORCE_BOOT_HEALTH_FAIL requires GS_HIL_CONTROL")
endif()
if(GS_HIL_FORCE_BOOT_HEALTH_FAIL AND GS_HIL_BUILD)
    message(FATAL_ERROR "Boot-health failure injection is supported only by the secure HIL-control profile")
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(gs_hw_m1_node)
