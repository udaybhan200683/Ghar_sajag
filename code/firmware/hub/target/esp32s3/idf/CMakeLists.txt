cmake_minimum_required(VERSION 3.16)
option(GS_S3_DIAGNOSTIC_ONLY "Build a development-only memory/radio diagnostic" OFF)
option(GS_HIL_COMMISSIONING "Enable the isolated S3 UART fixture commissioning adapter (test only)" OFF)
option(GS_HIL_C8_ACK_DROP "Enable one-shot postcommit ACK loss in the isolated S3 GS-114 fixture (test only)" OFF)
option(GS_HIL_C8_ACK_HOLD "Enable bounded exact-key postcommit ACK hold in the isolated S3 GS-114 fixture (test only)" OFF)
option(GS_HIL_SIGNED_FOTA "Enable exact-node authenticated signed-FOTA UART trigger in the isolated S3 fixture (test only)" OFF)
if(DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "This isolated Hub target requires IDF_TARGET=esp32s3")
endif()
set(IDF_TARGET esp32s3 CACHE STRING "Selected Hub chip")
if(GS_HIL_BUILD OR GS_HIL_CONTROL)
    message(FATAL_ERROR "S3 board-specific HIL GPIO/control qualification is pending")
endif()
if(GS_HIL_COMMISSIONING AND GS_S3_DIAGNOSTIC_ONLY)
    message(FATAL_ERROR "Fixture commissioning requires the complete S3 product composition")
endif()
if(GS_HIL_C8_ACK_DROP AND (NOT GS_HIL_COMMISSIONING OR GS_S3_DIAGNOSTIC_ONLY))
    message(FATAL_ERROR "GS_HIL_C8_ACK_DROP requires the complete S3 commissioning fixture profile")
endif()
if(GS_HIL_C8_ACK_HOLD AND (NOT GS_HIL_COMMISSIONING OR GS_S3_DIAGNOSTIC_ONLY OR GS_HIL_C8_ACK_DROP))
    message(FATAL_ERROR "GS_HIL_C8_ACK_HOLD requires the isolated S3 commissioning profile without ACK-drop")
endif()
if(GS_HIL_SIGNED_FOTA AND (NOT GS_HIL_COMMISSIONING OR GS_S3_DIAGNOSTIC_ONLY))
    message(FATAL_ERROR "GS_HIL_SIGNED_FOTA requires the isolated S3 commissioning fixture profile")
endif()
set(GS_HIL_BUILD OFF CACHE BOOL "Legacy insecure HIL composition" FORCE)
set(GS_HIL_CONTROL OFF CACHE BOOL "Legacy HIL controls" FORCE)
if(GS_S3_DIAGNOSTIC_ONLY)
    set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/diagnostic")
    set(COMPONENTS diagnostic)
else()
    # Reuse the complete existing product composition, including its C3 asset.
    set(EXTRA_COMPONENT_DIRS
        "${CMAKE_CURRENT_LIST_DIR}/../../esp32/idf/main"
        "${CMAKE_CURRENT_LIST_DIR}/s3_board"
        "${CMAKE_CURRENT_LIST_DIR}/s3_storage")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(gs_r1_s3_hub)
